OPÇÃO 1: Deploy Completo Automatizado (RECOMENDADO)
----------------------------------------------------
Um único comando que faz tudo:

  bash -c "$(curl -fsSL https://raw.githubusercontent.com/egarabini/Intellicare/staging/scripts/server/full_deploy.sh)"

Ou se o repositório já existe:

  cd /opt/intellicare
  bash scripts/server/full_deploy.sh


OPÇÃO 2: Passo a Passo
----------------------

# 1. Atualizar código do Git
cd /opt/intellicare
git remote set-url origin https://github.com/egarabini/Intellicare.git
git fetch origin
git reset --hard origin/staging

# 2. Configurar ambiente
bash scripts/server/setup_env.sh

# 3. Remover network antigo (se existir)
docker network rm intellicare_intellicare-network 2>/dev/null || true

# 4. Parar containers antigos
docker-compose -f docker-compose.full.yml down

# 5. Rebuild e start
docker-compose -f docker-compose.full.yml build
docker-compose -f docker-compose.full.yml up -d

# 6. Verificar status
docker ps


OPÇÃO 3: Clone Limpo (se /opt/intellicare não existe)
-----------------------------------------------------

cd /opt
rm -rf intellicare
git clone --branch staging --single-branch https://github.com/egarabini/Intellicare.git intellicare
cd intellicare
bash scripts/server/setup_env.sh
docker-compose -f docker-compose.full.yml build
docker-compose -f docker-compose.full.yml up -d


VERIFICAÇÃO
-----------

# Smoke test de todos os módulos
bash scripts/smoke_test.sh

# Ver logs de um container específico
docker logs intellicare-wanda
docker logs intellicare-florence

# Ver todos os logs
docker-compose -f docker-compose.full.yml logs -f


TROUBLESHOOTING
---------------

# Container reiniciando constantemente
docker logs intellicare-<nome-do-modulo>

# Rebuild específico de um módulo
docker-compose -f docker-compose.full.yml build intellicare-<modulo>
docker-compose -f docker-compose.full.yml up -d intellicare-<modulo>

# Ver variáveis de ambiente
cat .env
cat .env.full

# Entrar em um container para debug
docker exec -it intellicare-wanda bash
