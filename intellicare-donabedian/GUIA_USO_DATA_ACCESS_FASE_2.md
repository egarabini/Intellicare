# Exemplo 1: Criar novo Pilar via serviço
# -----------------------------------------

from uuid import UUID
from datetime import datetime
from sqlalchemy.orm import Session

# Import from data_access
from donabedian.data_access import OperationalDataAccess
from donabedian.models import Pillar


def create_pilar_service(
    session: Session,
    nome: str,
    descricao: str,
    tipo: str,
    actor_id: UUID,  # User creating this
) -> Pillar:
    """
    Serviço para criar novo pilar com auditoria.
    """
    
    # Instancia OperationalDataAccess
    op_dao = OperationalDataAccess(
        session=session,
        entity_class=Pillar,
        event_callback=None,  # FASE 2.2: Será substituído por EventPublisher
    )
    
    # Cria entidade
    pilar = Pillar(
        id=None,  # Will be auto-generated by DAO if needed
        nome=nome,
        descricao=descricao,
        tipo=tipo,
        ordem_exibicao=1,
        ativo=True,
        created_by=actor_id,  # Required!
    )
    
    # Usa DAO para criar (com auditoria)
    created = op_dao.create(
        entity=pilar,
        actor_id=actor_id,
        reason=f"Created pillar '{nome}' via API"  # Auditado em logs
    )
    
    # Retorna para API
    return created


# Exemplo 2: Atualizar Pilar via serviço
# ----------------------------------------

def update_pilar_service(
    session: Session,
    pilar_id: UUID,
    nome: str,
    ativo: bool,
    actor_id: UUID,
) -> Pillar:
    """
    Serviço para atualizar pilar com auditoria e versionamento.
    """
    
    op_dao = OperationalDataAccess(session, Pillar, event_callback=None)
    
    # Lê pilar existente
    pilar = op_dao.read(pilar_id)
    if not pilar:
        raise ValueError(f"Pilar {pilar_id} não encontrado")
    
    # Atualiza campos
    pilar.nome = nome
    pilar.ativo = ativo
    pilar.updated_by = actor_id  # Required!
    
    # Salva com DAO (incrementa rowversion automaticamente)
    updated = op_dao.update(
        entity=pilar,
        actor_id=actor_id,
        reason=f"Updated pilar: nome={nome}, ativo={ativo}"
    )
    
    return updated


# Exemplo 3: Deletar Pilar (soft delete)
# ----------------------------------------

def delete_pilar_service(
    session: Session,
    pilar_id: UUID,
    actor_id: UUID,
) -> bool:
    """
    Serviço para soft-deletar pilar (preserva auditoria).
    """
    
    op_dao = OperationalDataAccess(session, Pillar, event_callback=None)
    
    # Soft delete (seta valid_to = NOW())
    success = op_dao.delete(
        entity_id=pilar_id,
        actor_id=actor_id,
        reason="Pilar marca como inativo"
    )
    
    return success

def get_pilares_for_dashboard(

        session=session,

    pilares = analytics_dao.list(limit=100, offset=0)
    
    return pilares


# Exemplo 5: Aggregações para KPIs

from donabedian.models import Measurement


def get_measurement_statistics(
    session: Session,
) -> dict:
    """
    Calcula estatísticas de medições para relatórios.
    """
    
    analytics_dao = AnalyticsDataAccess(session, Measurement)
    
    # Aggregações
    stats = {
        "total_measurements": analytics_dao.count("id"),
        "valor_stats": analytics_dao.get_statistics("valor"),
        "by_status": analytics_dao.aggregate("status", "COUNT"),
    }
    
    return stats

@router.post("/")
async def criar_pilar(
    nome: str,
    descricao: str,
    tipo: str,
    session: Session = Depends(get_db_session),
    actor_id: UUID = Depends(get_current_user_id),
) -> dict:
    """
    POST /pilares
    
    Cria novo pilar com auditoria.
    """
    try:
        pilar = create_pilar_service(
            session=session,
            nome=nome,
            descricao=descricao,
            tipo=tipo,
            actor_id=actor_id,
        )
        
        session.commit()
        
        return {
            "id": str(pilar.id),
            "nome": pilar.nome,
            "created_by": str(pilar.created_by),
            "created_at": pilar.created_at.isoformat(),
        }
    
    except Exception as e:
        session.rollback()
        raise HTTPException(status_code=400, detail=str(e))


@router.put("/{pilar_id}")
async def atualizar_pilar(
    pilar_id: UUID,
    nome: str,
    ativo: bool,
    session: Session = Depends(get_db_session),
    actor_id: UUID = Depends(get_current_user_id),
) -> dict:
    """
    PUT /pilares/{pilar_id}
    
    Atualiza pilar com auditoria e versionamento.
    """
    try:
        pilar = update_pilar_service(
            session=session,
            pilar_id=pilar_id,
            nome=nome,
            ativo=ativo,
            actor_id=actor_id,
        )
        
        session.commit()
        
        return {
            "id": str(pilar.id),
            "nome": pilar.nome,
            "rowversion": pilar.rowversion,  # Show version to client
            "updated_by": str(pilar.updated_by),
            "updated_at": pilar.updated_at.isoformat(),
        }
    
    except ValueError as e:
        raise HTTPException(status_code=404, detail=str(e))
    except Exception as e:
        session.rollback()
        raise HTTPException(status_code=400, detail=str(e))


@router.delete("/{pilar_id}")
async def deletar_pilar(
    pilar_id: UUID,
    session: Session = Depends(get_db_session),
    actor_id: UUID = Depends(get_current_user_id),
) -> dict:
    """
    DELETE /pilares/{pilar_id}
    
    Soft-deleta pilar (preserva para auditoria).
    """
    try:
        success = delete_pilar_service(
            session=session,
            pilar_id=pilar_id,
            actor_id=actor_id,
        )
        
        if not success:
            raise HTTPException(status_code=404, detail="Pilar não encontrado")
        
        session.commit()
        
        return {"deleted": True, "id": str(pilar_id)}
    
    except Exception as e:
        session.rollback()
        raise HTTPException(status_code=400, detail=str(e))

# ERROR HANDLING: Versioning Conflict (Optimistic Locking)
# ============================================================================

"""
Se dois usuários tentam atualizar o mesmo pilar concorrentemente:

1. User A lê pilar, rowversion = 1
2. User B lê pilar, rowversion = 1
3. User A atualiza (rowversion → 2) OK
4. User B tenta atualizar com rowversion = 1
   → DAO detecta conflito
   → Lança StaleObjectError ou similar
   → API retorna 409 Conflict ao User B

Solução: User B refaz o READ (agora vê rowversion=2) e tenta UPDATE novamente.

Código do DAO (em FASE 1):
    if entity.rowversion != db_entity.rowversion:
        raise StaleObjectError(f"Versão conflito: esperado {db_entity.rowversion}, got {entity.rowversion}")
"""

"""
A cada CREATE/UPDATE/DELETE, os campos de auditoria são preenchidos:
- created_by: UUID do usuário que criou
- created_at: timestamp de criação
- updated_by: UUID do usuário que atualizou
- updated_at: timestamp de atualização
- valid_to: timestamp de soft-delete (se deletado)

Para ver histórico de quem mexeu num pilar:

SELECT 
    id, 
    nome,
    created_by,
    created_at,
    updated_by,
    updated_at,
    valid_to,
    rowversion
FROM donabedian_operacional.pilares
WHERE id = '...'
ORDER BY updated_at DESC;

Resultado:
- Se valid_to IS NULL: pilar ainda está ativo
- Se valid_to IS NOT NULL: pilar foi deletado em valid_to
- rowversion mostra quantas vezes foi atualizado
"""
